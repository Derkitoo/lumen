// Hook pour la persistance des données dans le navigateur
function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.warn("Erreur de lecture du localStorage", error);
      return initialValue;
    }
  });

  const setValue = (value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.warn("Erreur d'écriture dans le localStorage", error);
    }
  };

  return [storedValue, setValue];
}

const BreathePhase = ({ onComplete, onCancel, standalone = false }) => {
  const [phase, setPhase] = useState('inspire');
  const [timeLeft, setTimeLeft] = useState(60); 
  
  useEffect(() => {
    let breatheInterval;
    let timerInterval;

    const runBreathingCycle = () => {
      setPhase('inspire');
      setTimeout(() => setPhase('expire'), 5000);
    };

    runBreathingCycle();
    breatheInterval = setInterval(runBreathingCycle, 10000);

    timerInterval = setInterval(() => {
      setTimeLeft((prev) => {
        if (prev <= 1) {
          clearInterval(breatheInterval);
          clearInterval(timerInterval);
          onComplete(); 
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => {
      clearInterval(breatheInterval);
      clearInterval(timerInterval);
    };
  }, [onComplete]);

  return (
    <div className="flex flex-col h-full animate-fade-in relative z-10 bg-slate-900">
      <Header title="Cohérence cardiaque" subtitle="Apaisement du système nerveux" onBack={onCancel} />
      
      <div className="flex-1 flex flex-col items-center justify-center gap-12">
        <div className="text-center">
          <h2 className="text-2xl font-bold text-slate-100 mb-2">Respirez profondément</h2>
          <p className="text-slate-400 text-sm">Synchronisez votre souffle avec le cercle.</p>
        </div>

        <div className="relative flex items-center justify-center w-64 h-64">
          <div 
            className="absolute bg-teal-500/10 rounded-full transition-all ease-in-out"
            style={{ width: phase === 'inspire' ? '100%' : '40%', height: phase === 'inspire' ? '100%' : '40%', transitionDuration: '5000ms' }}
          />
          <div 
            className="absolute bg-teal-500/20 rounded-full transition-all ease-in-out"
            style={{ width: phase === 'inspire' ? '80%' : '30%', height: phase === 'inspire' ? '80%' : '30%', transitionDuration: '5000ms' }}

            <Wind size={28} className={`mb-1 transition-colors duration-1000 ${phase === 'inspire' ? 'text-teal-400' : 'text-emerald-400'}`} />
            <span className="text-sm font-semibold tracking-widest text-slate-300 uppercase">
              {phase === 'inspire' ? 'Inspirez' : 'Expirez'}

          </div>
        </div>

        <div className="w-full max-w-xs mt-4">
          <div className="flex justify-between text-xs font-medium text-slate-500 mb-2 uppercase tracking-wider">
            <span>Progression</span>
            <span>00:{timeLeft.toString().padStart(2, '0')}</span>

            />
          </div>
        </div>
      </div>
      
      {!standalone && (
        <div className="py-6 pb-8">
          <Button variant="ghost" onClick={onComplete} className="text-sm text-slate-500 hover:text-slate-300">

const TimerPhase = ({ durationMinutes, onFinish, onAbort }) => {

  const [showConfirm, setShowConfirm] = useState(false);

    const timer = setInterval(() => {
      setTimeLeft(prev => {
        if (prev <= 1) {
          clearInterval(timer);
          onFinish();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
    return () => clearInterval(timer);
  }, [onFinish]);

  const formatTime = (seconds) => {
    const m = Math.floor(seconds / 60);
    const s = seconds % 60;
    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
  };

          </>
        )}

        {step === 3 && currentFamily && (
          <>
            <div className="inline-flex p-3 rounded-xl bg-slate-800 text-slate-300 mb-4 w-fit border border-slate-700">
              <Activity size={24} />
            </div>
            <h2 className="text-xl font-bold text-slate-100 mb-2">Où se loge cette émotion ?</h2>
            <p className="text-slate-400 text-sm mb-6">L'émotion s'exprime d'abord dans le corps. Localisez-la pour vous en défusionner.</p>
            <div className="grid grid-cols-2 gap-3">
              {SOMATIC_SENSATIONS.map((sens, idx) => (
                <button 
                  key={idx} 
                  onClick={() => handleNext('somatic', sens)}
                  className="bg-slate-800/50 border border-slate-700 hover:border-indigo-500/50 p-4 rounded-xl text-xs font-medium text-slate-300 text-left hover:bg-slate-700/50 transition-all"
                >
                  {sens}
                </button>
              ))}
            </div>
          </>
        )}

        {step === 4 && currentFamily && (
          <>
            <div className="inline-flex p-3 rounded-xl bg-indigo-500/10 text-indigo-400 mb-4 w-fit">
              <Search size={24} />
            </div>
            <h2 className="text-xl font-bold text-slate-100 mb-2">Le besoin caché</h2>
            <p className="text-slate-400 text-sm mb-6">Toute émotion douloureuse est le signal d'alarme d'un besoin profond qui n'est pas nourri. Quel est le vôtre ?</p>
            <div className="flex flex-col gap-3">
              {currentFamily.needs.map((need, idx) => (
                <Button key={idx} variant="outline" onClick={() => handleNext('need', need)} className="py-4 justify-between px-6 text-sm hover:border-indigo-500/50 group">
                  <span>Besoin de <strong className="text-indigo-400 group-hover:text-indigo-300 font-bold ml-1">{need}</strong></span>
                  <ChevronRight size={16} className="text-slate-600 group-hover:text-indigo-400" />
                </Button>
              ))}
            </div>
          </>
        )}

        {step === 5 && currentFamily && (
          <div className="flex flex-col h-full">
            <div className="inline-flex p-3 rounded-xl bg-teal-500/10 text-teal-400 mb-4 w-fit">
              <Target size={24} />
            </div>
            <h2 className="text-xl font-bold text-slate-100 mb-2">La Dichotomie du Contrôle</h2>
            <p className="text-slate-400 text-sm mb-4">Vous ne pouvez pas forcer les autres à nourrir votre <span className="font-bold text-indigo-400">besoin de {analysis.need?.toLowerCase()}</span>. Mais que pouvez-vous faire <span className="underline decoration-teal-500 underline-offset-2">vous-même</span>, aujourd'hui, pour le nourrir ?</p>
            
            <textarea 
              value={analysis.action}
              onChange={(e) => setAnalysis(prev => ({ ...prev, action: e.target.value }))}
              placeholder="Ex: Je vais prendre un bain chaud, écrire dans mon journal, appeler un(e) ami(e) de confiance, mettre des limites claires..."
              className="w-full flex-1 min-h-[120px] bg-slate-900 border border-slate-700 rounded-xl p-4 text-slate-200 text-sm focus:outline-none focus:border-teal-500 mb-6 resize-none shadow-inner"
            />
            
            <Button onClick={handleFinish} disabled={!analysis.action.trim()} variant="success" className="py-4 mt-auto">
              Valider l'action et terminer
            </Button>
          </div>
        )}
      </div>
    </div>
  );
};

// --- MAIN APP CONTROLLER ---

const App = () => {
  // Global State
  const [journal, setJournal] = useLocalStorage('sos-detachement-journal-v2', []);
  const [settings, setSettings] = useLocalStorage('sos-detachement-settings-v2', { timerDuration: 45 });
  
  // Navigation State
  const [currentTab, setCurrentTab] = useState('home'); // home, library, tools, journal
  
  // Flow States
  const [activeFlow, setActiveFlow] = useState(null); // 'sos_breathe', 'sos_reframe', 'sos_timer', 'tool_breathe', 'tool_reframe', 'tool_timer', 'emotion_analyzer'
  const [activeCrisis, setActiveCrisis] = useState(null);

  // Global Styling Setup
  useEffect(() => {
    const style = document.createElement('style');
    style.textContent = `
      @import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700;800&display=swap');
      body { 
        background-color: #020617; /* slate-950 */
        color: #f8fafc; 
        font-family: 'Inter', sans-serif; 
        -webkit-tap-highlight-color: transparent;
      }
      .hide-scrollbar::-webkit-scrollbar { display: none; }
      .hide-scrollbar { -ms-overflow-style: none; scrollbar-width: none; }
      @keyframes fade-in { from { opacity: 0; } to { opacity: 1; } }
      @keyframes fade-in-up { from { opacity: 0; transform: translateY(10px); } to { opacity: 1; transform: translateY(0); } }
      .animate-fade-in { animation: fade-in 0.4s ease-out forwards; }
      .animate-fade-in-up { animation: fade-in-up 0.4s cubic-bezier(0.16, 1, 0.3, 1) forwards; }
    `;
    document.head.appendChild(style);
    return () => document.head.removeChild(style);
  }, []);

  // Handlers
  const startSOSFlow = (crisisId) => {
    if (crisisId === 'emotion_analyzer') {
      setActiveFlow('emotion_analyzer');
    } else {
      setActiveCrisis(crisisId);
      setActiveFlow('sos_breathe');
    }
  };
  
  const finishSOSFlow = () => {
    const newEntry = { id: Date.now(), type: 'sos', date: new Date().toISOString(), crisisId: activeCrisis };
    setJournal([newEntry, ...journal]);
    setActiveFlow(null);
    setActiveCrisis(null);
  };

  const handleMoodCheck = (mood) => {
    const newEntry = { id: Date.now(), type: 'mood', date: new Date().toISOString(), mood };
    setJournal([newEntry, ...journal]);
  };

  const addJournalEntry = (entry) => {
    setJournal([{ id: Date.now(), date: new Date().toISOString(), ...entry }, ...journal]);
  };

  // Rendering logic based on active flow
  const renderFlow = () => {
    switch (activeFlow) {
      // SOS FLOW
      case 'sos_breathe': return <BreathePhase onComplete={() => setActiveFlow('sos_reframe')} onCancel={() => setActiveFlow(null)} />;
      case 'sos_reframe': return <ReframePhase onComplete={() => setActiveFlow('sos_timer')} onCancel={() => setActiveFlow(null)} />;
      case 'sos_timer': return <TimerPhase durationMinutes={settings.timerDuration} onFinish={finishSOSFlow} onAbort={() => setActiveFlow(null)} />;
      
      // EMOTION ANALYZER FLOW
      case 'emotion_analyzer': 
        return <EmotionAnalyzerFlow 
          onComplete={(analysis) => {
            const newEntry = { id: Date.now(), type: 'emotion_analysis', date: new Date().toISOString(), analysis };
            setJournal([newEntry, ...journal]);
            setActiveFlow(null);
          }} 
          onCancel={() => setActiveFlow(null)} 
        />;

      // STANDALONE TOOLS FLOW
      case 'tool_breathe': return <BreathePhase onComplete={() => setActiveFlow(null)} onCancel={() => setActiveFlow(null)} standalone={true} />;
      case 'tool_reframe': return <ReframePhase onComplete={() => setActiveFlow(null)} onCancel={() => setActiveFlow(null)} />;
      case 'tool_timer': return <TimerPhase durationMinutes={settings.timerDuration} onFinish={() => setActiveFlow(null)} onAbort={() => setActiveFlow(null)} />;
      
      default: return null;
    }
  };

  return (
    <div className="min-h-screen flex justify-center bg-black">
      <main className="w-full max-w-md bg-slate-950 relative flex flex-col h-[100dvh] overflow-hidden">
        
        {/* VIEW PORT */}
        <div className="flex-1 px-5 pt-6 pb-0 relative overflow-hidden">
          {!activeFlow ? (
            <>
              {currentTab === 'home' && <DashboardView onStartSOS={startSOSFlow} onMoodCheck={handleMoodCheck} journal={journal} />}
              {currentTab === 'library' && <LibraryView />}
              {currentTab === 'tools' && <ToolsView startSpecificTool={(tool) => setActiveFlow(`tool_${tool}`)} />}
              {currentTab === 'journal' && <JournalView journal={journal} addJournalEntry={addJournalEntry} clearJournal={() => setJournal([])} />}
            </>
          ) : (
            renderFlow()
          )}
        </div>

        {/* BOTTOM NAVIGATION BAR */}
        {!activeFlow && (
          <div className="absolute bottom-0 left-0 right-0 p-4 pb-6 bg-gradient-to-t from-slate-950 via-slate-950 to-transparent z-40">
            <div className="flex justify-between items-center bg-slate-800/80 backdrop-blur-xl border border-slate-700/50 rounded-2xl p-2 shadow-2xl">
              {[
                { id: 'home', icon: Home, label: "Accueil" },
                { id: 'library', icon: BookOpen, label: "Comprendre" },
                { id: 'tools', icon: Compass, label: "Pratiquer" },
                { id: 'journal', icon: Calendar, label: "Espace" }
              ].map(tab => {
                const isActive = currentTab === tab.id;
                const Icon = tab.icon;
                return (
                  <button 
                    key={tab.id}
                    onClick={() => setCurrentTab(tab.id)}
                    className={`flex flex-col items-center gap-1 p-2 flex-1 rounded-xl transition-all duration-300 ${isActive ? 'text-indigo-400 bg-indigo-500/10' : 'text-slate-500 hover:text-slate-300'}`}
                  >
                    <Icon size={20} className={isActive ? 'animate-fade-in-up' : ''} />
                    <span className="text-[9px] font-semibold uppercase tracking-wider">{tab.label}</span>
                  </button>
                );
              })}
            </div>
          </div>
        )}
      </main>
    </div>
  );
};

export default App;
